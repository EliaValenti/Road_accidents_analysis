INDICI per ISTAT:

FREQ= frequenza --[A] annuale
REF_AREA= comuni --- -> ID dei comuni
DATA_TYPE = indicatore -- [KILLINJ] morti e feriti
                        [ROADACC] incidenti stradali con lesioni alle persone
RESULT= esito -- [M] morto [F] ferito [9] totale --da sostituire 
TIME_PERIOD = Anno di osservazione
OBS_VALUE= Valori osservati (observed value)
